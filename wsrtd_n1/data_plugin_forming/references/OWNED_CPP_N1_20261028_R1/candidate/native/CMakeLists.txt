cmake_minimum_required(VERSION 3.20)
project(WSRTD_OWNED_R5B LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../dependencies")
set(ADK "${DEPS}/ADK/ADK/include")
set(SQLITE "${DEPS}/sqlite/sqlite-amalgamation-3540100")
add_library(sqlite STATIC "${SQLITE}/sqlite3.c")
target_include_directories(sqlite PUBLIC "${SQLITE}")
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0)
add_library(owned_reader STATIC owned_reader.cpp)
target_include_directories(owned_reader PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(owned_reader PUBLIC sqlite)
add_library(WSRTD_Owned_N1_DATABASE_BOUND SHARED owned_plugin.cpp)
target_include_directories(WSRTD_Owned_N1_DATABASE_BOUND PRIVATE "${ADK}")
target_link_libraries(WSRTD_Owned_N1_DATABASE_BOUND PRIVATE owned_reader)
target_compile_definitions(WSRTD_Owned_N1_DATABASE_BOUND PRIVATE _USRDLL=1 _MBCS)
target_compile_options(WSRTD_Owned_N1_DATABASE_BOUND PRIVATE /W4 /EHsc /analyze)
add_executable(n1_probe n1_probe.cpp)
target_include_directories(n1_probe PRIVATE "${ADK}")
target_compile_definitions(n1_probe PRIVATE _MBCS)
target_compile_options(owned_reader PRIVATE /W4 /EHsc /analyze)
