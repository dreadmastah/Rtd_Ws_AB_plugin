cmake_minimum_required(VERSION 3.20)
project(WSRTD_FORMING_R3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 only")
endif()
add_library(WSRTD_Data_Forming_R3 SHARED plugin.cpp)
target_include_directories(WSRTD_Data_Forming_R3 PRIVATE ../dependencies/adk)
target_compile_definitions(WSRTD_Data_Forming_R3 PRIVATE _USRDLL=1 _MBCS)
target_compile_options(WSRTD_Data_Forming_R3 PRIVATE /W4 /EHsc)
target_link_libraries(WSRTD_Data_Forming_R3 PRIVATE user32)
add_executable(native_probe ../tests/native_probe.cpp)
target_include_directories(native_probe PRIVATE ../dependencies/adk)
target_compile_definitions(native_probe PRIVATE _MBCS)
target_link_libraries(native_probe PRIVATE user32 psapi)
